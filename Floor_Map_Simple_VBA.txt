'======================================================================
'  FLOOR MAP (SIMPLE) - MACROS
'
'  HOW TO PUT THIS IN (once):
'   1. Open Floor_Map_Simple.xlsx. File > Save As > "Excel Macro-Enabled
'      Workbook (*.xlsm)". Use the .xlsm from now on.
'   2. Press Alt + F11.
'   3. On the left, double-click the sheet that says (Map).
'   4. Copy EVERYTHING below the line of stars and paste it in.
'      (Pasted an older version before? Click in the code, Ctrl+A, Delete,
'       then paste this one.)
'   5. Close the window. Click any square on the Map - the buttons appear.
'   6. Ctrl + S to save.
'
'  WHAT YOU GET:
'   * MOVE    - double-click a pallet, then click where it goes
'               (click another pallet to swap, Esc or Clear to cancel)
'   * PUT AWAY- double-click an empty square and pick from a list
'   * Buttons - Find space / Find item / Low stock / Undo / Clear / Rebuild map
'   * Change aisle sizes on the Layout sheet, then press Rebuild map
'   * Every change is written to the Moves sheet, and Undo can reverse it
'======================================================================
'**********************************************************************
Option Explicit

Private Const BAY_ROW As Long = 4           ' loading bay row on the Map
Private Const MAX_UNDO As Long = 50

Private mHeld As Range                      ' pallet picked up to move
Private mBusy As Boolean                    ' true while a macro changes squares
Private mOldAddr As String                  ' last single cell clicked...
Private mOldVal As String                   ' ...and what was in it
Private mJust As Single                     ' time of the last move (double-click guard)
Private mUndo As Collection                 ' undo steps: "addr=value<TAB>addr=value..."
Private mHi As Collection                   ' squares with a pink highlight

Private sqId() As String, sqZone() As String
Private sqRow() As Long, sqCol() As Long, sqDepth() As Long, sqN As Long

' Used by Rebuild map
Private Const TOP_ROW As Long = 6           ' first row of aisles on the Map
Private Const WALK As Long = 3              ' walkway width in squares
Private Const LEG_COLS As Long = 10         ' legend width in squares
Private Const SQ_PX As Long = 28            ' square size in pixels

Private gBottom As Long, gWalkS As Long, gWalkE As Long, gRightS As Long
Private gMapLast As Long, gLegend As Long, gLast As Long, gFire As Long
Private zCode() As String, zN As Long

' The "Pallets on map" formula on the Stock sheet (counts the whole Map)
Private Const COUNT_FORMULA As String = _
    "=IF(TRIM([@[Item code]])="""","""",COUNTIF(Map!$A$6:$ZZ$2000,TRIM([@[Item code]])))"
Private mStockChecked As Boolean


'======================================================================
'  EVENTS
'======================================================================
Private Sub Worksheet_Activate()
    ' counts on the Stock sheet only work when Excel calculates by itself
    If Application.Calculation <> xlCalculationAutomatic Then Application.Calculation = xlCalculationAutomatic
    EnsureButtons
End Sub

Private Sub Worksheet_SelectionChange(ByVal Target As Range)
    Dim i As Long
    If mBusy Then Exit Sub
    EnsureButtons
    If Target.CountLarge <> 1 Then Exit Sub
    mOldAddr = Target.Address
    mOldVal = CellText(Target)
    i = SqIndex(Target)
    If i = 0 Then Exit Sub
    If Not mHeld Is Nothing Then
        If Target.Address <> mHeld.Address Then DoMove mHeld, Target
    Else
        ShowInfo i, Target
    End If
End Sub

Private Sub Worksheet_BeforeDoubleClick(ByVal Target As Range, Cancel As Boolean)
    Dim i As Long
    If Target.CountLarge <> 1 Then Exit Sub
    i = SqIndex(Target)
    If i = 0 Then Exit Sub
    Cancel = True                           ' don't start typing in the cell
    If Abs(Timer - mJust) < 1 Then Exit Sub ' the click just finished a move
    If Not mHeld Is Nothing Then
        If Target.Address = mHeld.Address Then
            CancelMove
        Else
            DoMove mHeld, Target
        End If
    ElseIf CellText(Target) <> "" Then
        PickUp Target
    Else
        PutAway Target, i
    End If
End Sub

' Picked from the dropdown, typed, or deleted by hand: log it and allow Undo
Private Sub Worksheet_Change(ByVal Target As Range)
    Dim c As Range, i As Long, newV As String
    If mBusy Then Exit Sub
    If Target.CountLarge > 500 Then Exit Sub
    For Each c In Target.Cells
        i = SqIndex(c)
        If i > 0 Then
            newV = CellText(c)
            If Target.CountLarge = 1 And c.Address = mOldAddr Then
                If newV <> mOldVal Then
                    PushUndo c.Address & "=" & mOldVal
                    If newV = "" Then
                        LogIt "Removed", mOldVal, sqId(i), ""
                    ElseIf mOldVal = "" Then
                        LogIt "Put away", newV, "", sqId(i)
                    Else
                        LogIt "Changed", newV, sqId(i) & " (was " & mOldVal & ")", sqId(i)
                    End If
                End If
            Else
                LogIt IIf(newV = "", "Removed", "Edited"), newV, "", sqId(i)
            End If
        End If
    Next c
    If Target.CountLarge = 1 Then mOldVal = CellText(Target)
End Sub

' Find space: pick an item and how many pallets; shows the nearest free
' squares to the loading bay in that item's zone and offers to fill them.
Public Sub FindSpace()
    Dim code As String, zone As String, qty As Variant, n As Long
    Dim inZ() As Long, outZ() As Long, nIn As Long, nOut As Long, i As Long, k As Long
    Dim useIn As Long, useOut As Long, msg As String, undo As String, c As Range
    On Error GoTo Fail
    CancelMove
    ClearHighlights
    code = PickItem("FIND SPACE" & vbLf & "Which item has arrived?", "")
    If code = "" Then Exit Sub
    zone = ItemZone(code)
    qty = Application.InputBox("How many pallets of " & code & "?", "Find space", 1, Type:=1)
    If VarType(qty) = vbBoolean Then Exit Sub
    n = CLng(qty)
    If n < 1 Then Exit Sub

    LoadSquares
    ReDim inZ(1 To sqN): ReDim outZ(1 To sqN)
    For i = 1 To sqN
        If CellText(Me.Cells(sqRow(i), sqCol(i))) = "" Then
            If zone <> "" And StrComp(sqZone(i), zone, vbTextCompare) = 0 Then
                nIn = nIn + 1: inZ(nIn) = i
            Else
                nOut = nOut + 1: outZ(nOut) = i
            End If
        End If
    Next i
    SortNearestBay inZ, nIn
    SortNearestBay outZ, nOut
    useIn = n: If useIn > nIn Then useIn = nIn
    useOut = n - useIn: If useOut > nOut Then useOut = nOut

    For k = 1 To useIn
        Highlight Me.Cells(sqRow(inZ(k)), sqCol(inZ(k)))
    Next k
    For k = 1 To useOut
        Highlight Me.Cells(sqRow(outZ(k)), sqCol(outZ(k)))
    Next k
    If useIn + useOut > 0 Then
        Application.Goto Me.Cells(sqRow(IIf(useIn > 0, inZ(1), outZ(1))), sqCol(IIf(useIn > 0, inZ(1), outZ(1)))), False
    End If

    msg = code & ": " & n & " pallet" & IIf(n = 1, "", "s") & vbLf & vbLf
    If useIn > 0 Then msg = msg & useIn & " free in " & ZoneName(zone) & " (pink squares, nearest the bay)." & vbLf
    If useIn < n Then
        msg = msg & vbLf & ZoneName(zone) & " is " & IIf(nIn = 0, "FULL.", "short by " & (n - useIn) & ".") & vbLf
        If useOut > 0 Then msg = msg & useOut & " pink square" & IIf(useOut = 1, " is", "s are") & " in OTHER zones." & vbLf
        If useIn + useOut < n Then msg = msg & (n - useIn - useOut) & " pallet(s) won't fit anywhere." & vbLf
    End If
    If useIn + useOut = 0 Then
        MsgBox msg, vbExclamation, "Find space"
        Exit Sub
    End If
    If MsgBox(msg & vbLf & "Put " & code & " in the pink squares now?", vbYesNo + vbQuestion, "Find space") = vbYes Then
        mBusy = True
        For k = 1 To useIn + useOut
            If k <= useIn Then i = inZ(k) Else i = outZ(k - useIn)
            Set c = Me.Cells(sqRow(i), sqCol(i))
            undo = undo & c.Address & "=" & vbTab
            c.Value = code
            LogIt "Put away", code, "(Find space)", sqId(i)
        Next k
        mBusy = False
        PushUndo undo
        SetMsg "Done: " & (useIn + useOut) & " pallet(s) of " & code & " put away (pink). Press Clear to remove the pink.", RGB(0, 132, 61)
    Else
        ClearHighlights
    End If
    Exit Sub
Fail:
    mBusy = False
    MsgBox "Find space stopped: " & Err.Description, vbExclamation
End Sub

' Find item: pink border round every pallet of one item
Public Sub FindItem()
    Dim code As String, i As Long, n As Long, first As Range
    CancelMove
    ClearHighlights
    code = PickItem("FIND ITEM" & vbLf & "Which item are you looking for?", "")
    If code = "" Then Exit Sub
    LoadSquares
    For i = 1 To sqN
        If StrComp(CellText(Me.Cells(sqRow(i), sqCol(i))), code, vbTextCompare) = 0 Then
            Highlight Me.Cells(sqRow(i), sqCol(i))
            If first Is Nothing Then Set first = Me.Cells(sqRow(i), sqCol(i))
            n = n + 1
        End If
    Next i
    If n = 0 Then
        SetMsg code & " is not on the map.", RGB(192, 0, 0)
    Else
        Application.Goto first, False
        SetMsg code & ": " & n & " pallet" & IIf(n = 1, "", "s") & " - shown in pink. Press Clear when done.", RGB(230, 0, 126)
    End If
End Sub

' Low stock: list of items marked LOW on the Stock sheet
Public Sub LowStock()
    Dim lo As ListObject, i As Long, msg As String, n As Long
    Set lo = StockT
    If Not lo.DataBodyRange Is Nothing Then
        For i = 1 To lo.ListRows.Count
            If SafeText(lo.ListColumns("Status").DataBodyRange.Cells(i).Value) = "LOW" Then
                n = n + 1
                msg = msg & "   " & SafeText(lo.ListColumns("Item code").DataBodyRange.Cells(i).Value) & "  -  " & _
                      SafeText(lo.ListColumns("Description").DataBodyRange.Cells(i).Value) & "  (" & _
                      SafeText(lo.ListColumns("Pallets on map").DataBodyRange.Cells(i).Value) & " pallets)" & vbLf
            End If
        Next i
    End If
    If n = 0 Then
        MsgBox "Nothing is low. All good!", vbInformation, "Low stock"
    Else
        MsgBox n & " item" & IIf(n = 1, " is", "s are") & " LOW:" & vbLf & vbLf & msg, vbExclamation, "Low stock"
    End If
End Sub

' Undo the last change (move, swap, put away, remove, Find space)
Public Sub UndoLast()
    Dim parts() As String, i As Long, p As Long, addr As String, v As String
    CancelMove
    ClearHighlights
    If mUndo Is Nothing Then Set mUndo = New Collection
    If mUndo.Count = 0 Then
        SetMsg "Nothing to undo.", RGB(128, 128, 128)
        Exit Sub
    End If
    parts = Split(mUndo(mUndo.Count), vbTab)
    mUndo.Remove mUndo.Count
    mBusy = True
    For i = UBound(parts) To 0 Step -1
        p = InStr(parts(i), "=")
        If p > 0 Then
            addr = Left$(parts(i), p - 1)
            v = Mid$(parts(i), p + 1)
            LogIt "Undo", IIf(v = "", CellText(Me.Range(addr)), v), "", SqIdAt(Me.Range(addr))
            If v = "" Then Me.Range(addr).ClearContents Else Me.Range(addr).Value = v
            Highlight Me.Range(addr)
        End If
    Next i
    mBusy = False
    SetMsg "Undone (changed squares shown in pink). Press Clear to remove the pink.", RGB(0, 112, 192)
End Sub

' Clear: cancel a move and remove pink highlights
Public Sub ClearAll()
    CancelMove
    ClearHighlights
    SetMsg "", 0
End Sub

' Also called by the Esc key while a pallet is picked up
Public Sub CancelMove()
    If Not mHeld Is Nothing Then
        Grid mHeld
        Set mHeld = Nothing
        SetMsg "", 0
    End If
    Application.OnKey "{ESC}"
End Sub


'======================================================================
'  MOVING
'======================================================================
Private Sub PickUp(c As Range)
    ClearHighlights
    Set mHeld = c
    Box c, vbBlack
    Application.OnKey "{ESC}", Me.CodeName & ".CancelMove"
    SetMsg "MOVING " & CellText(c) & " from " & SqIdAt(c) & "  -  click where it goes  (Esc to cancel)", vbBlack
End Sub

Private Sub DoMove(src As Range, dst As Range)
    Dim a As String, b As String, ia As Long, ib As Long, msg As String
    a = CellText(src): b = CellText(dst)
    ia = SqIndex(src): ib = SqIndex(dst)
    msg = ZoneWarning(a, sqZone(ib), sqId(ib))
    If b <> "" Then msg = msg & ZoneWarning(b, sqZone(ia), sqId(ia))
    If msg <> "" Then
        If MsgBox(msg & vbLf & "Put it there anyway?", vbYesNo + vbExclamation + vbDefaultButton2, _
                  "Different zone") = vbNo Then
            CancelMove
            Exit Sub
        End If
    End If
    mBusy = True
    dst.Value = a
    If b = "" Then src.ClearContents Else src.Value = b
    mBusy = False
    Grid src
    Set mHeld = Nothing
    Application.OnKey "{ESC}"
    PushUndo src.Address & "=" & a & vbTab & dst.Address & "=" & b
    If b = "" Then
        LogIt "Moved", a, sqId(ia), sqId(ib)
        SetMsg "Moved " & a & " from " & sqId(ia) & " to " & sqId(ib) & ".", RGB(0, 132, 61)
    Else
        LogIt "Swapped", a, sqId(ia), sqId(ib)
        LogIt "Swapped", b, sqId(ib), sqId(ia)
        SetMsg "Swapped " & a & " and " & b & ".", RGB(0, 132, 61)
    End If
    mOldAddr = dst.Address
    mOldVal = CellText(dst)
    mJust = Timer
End Sub

' Double-click on an empty square: pick an item for it from a list
Private Sub PutAway(c As Range, ByVal i As Long)
    Dim code As String
    code = PickItem("PUT AWAY in " & sqId(i) & " (" & ZoneName(sqZone(i)) & ")", sqZone(i))
    If code = "" Then Exit Sub
    If ZoneWarning(code, sqZone(i), sqId(i)) <> "" Then
        If MsgBox(ZoneWarning(code, sqZone(i), sqId(i)) & vbLf & "Put it there anyway?", _
                  vbYesNo + vbExclamation + vbDefaultButton2, "Different zone") = vbNo Then Exit Sub
    End If
    mBusy = True
    c.Value = code
    mBusy = False
    PushUndo c.Address & "="
    LogIt "Put away", code, "", sqId(i)
    mOldVal = code
    SetMsg "Put away " & code & " in " & sqId(i) & ".", RGB(0, 132, 61)
End Sub

Private Function ZoneWarning(ByVal item As String, ByVal zone As String, ByVal id As String) As String
    Dim iz As String
    If item = "" Or zone = "" Then Exit Function
    iz = ItemZone(item)
    If iz = "" Then Exit Function
    If StrComp(iz, zone, vbTextCompare) <> 0 Then
        ZoneWarning = item & " belongs in " & ZoneName(iz) & ", but " & id & " is " & ZoneName(zone) & "." & vbLf
    End If
End Function


'======================================================================
'  LOOK-UPS
'======================================================================
Private Sub LoadSquares()
    Dim ws As Worksheet, v As Variant, i As Long, n As Long
    If sqN > 0 Then Exit Sub
    Set ws = ThisWorkbook.Worksheets("Squares")
    n = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row - 1
    If n < 1 Then Exit Sub
    v = ws.Range("A2").Resize(n, 6).Value
    ReDim sqId(1 To n): ReDim sqZone(1 To n): ReDim sqRow(1 To n): ReDim sqCol(1 To n): ReDim sqDepth(1 To n)
    For i = 1 To n
        sqId(i) = CStr(v(i, 1))
        sqZone(i) = CStr(v(i, 3))
        sqRow(i) = CLng(v(i, 4))
        sqCol(i) = CLng(v(i, 5))
        sqDepth(i) = CLng(v(i, 6))
    Next i
    sqN = n
End Sub

Private Function SqIndex(c As Range) As Long
    Dim i As Long
    LoadSquares
    For i = 1 To sqN
        If sqRow(i) = c.Row Then
            If sqCol(i) = c.Column Then
                SqIndex = i
                Exit Function
            End If
        End If
    Next i
End Function

Private Function SqIdAt(c As Range) As String
    Dim i As Long
    i = SqIndex(c)
    If i > 0 Then SqIdAt = sqId(i)
End Function

' Nearest the loading bay first: rows from the bay + distance from the walkway
Private Sub SortNearestBay(idx() As Long, ByVal n As Long)
    Dim i As Long, j As Long, t As Long
    For i = 2 To n
        t = idx(i)
        j = i - 1
        Do While j >= 1
            If NearKey(idx(j)) <= NearKey(t) Then Exit Do
            idx(j + 1) = idx(j)
            j = j - 1
        Loop
        idx(j + 1) = t
    Next i
End Sub

Private Function NearKey(ByVal i As Long) As Double
    NearKey = (sqRow(i) - BAY_ROW + sqDepth(i)) * 1000 + sqRow(i)
End Function

Private Function StockT() As ListObject
    Set StockT = ThisWorkbook.Worksheets("Stock").ListObjects("tblStock")
End Function

Private Function ItemRow(ByVal code As String) As Long
    Dim lo As ListObject, i As Long
    Set lo = StockT
    If lo.DataBodyRange Is Nothing Or code = "" Then Exit Function
    For i = 1 To lo.ListRows.Count
        If StrComp(SafeText(lo.ListColumns("Item code").DataBodyRange.Cells(i).Value), code, vbTextCompare) = 0 Then
            ItemRow = i
            Exit Function
        End If
    Next i
End Function

Private Function ItemZone(ByVal code As String) As String
    Dim r As Long
    r = ItemRow(code)
    If r > 0 Then ItemZone = SafeText(StockT.ListColumns("Zone").DataBodyRange.Cells(r).Value)
End Function

Private Function ItemDesc(ByVal code As String) As String
    Dim r As Long
    r = ItemRow(code)
    If r > 0 Then ItemDesc = SafeText(StockT.ListColumns("Description").DataBodyRange.Cells(r).Value)
End Function

' The name typed on the Stock sheet for a zone code (falls back to the code)
Private Function ZoneName(ByVal code As String) As String
    ZoneName = code
    If code = "" Then ZoneName = "no zone": Exit Function
    On Error Resume Next
    ZoneName = SafeText(ThisWorkbook.Names("zn_" & code).RefersToRange.Value)
    If ZoneName = "" Then ZoneName = code
End Function

' Numbered list of items; returns the chosen item code or ""
Private Function PickItem(ByVal title As String, ByVal zoneFilter As String) As String
    Dim lo As ListObject, i As Long, n As Long, codes() As String, lst As String
    Dim code As String, z As String, ans As String
    Set lo = StockT
    If Not lo.DataBodyRange Is Nothing Then
        ReDim codes(1 To lo.ListRows.Count)
        For i = 1 To lo.ListRows.Count
            code = SafeText(lo.ListColumns("Item code").DataBodyRange.Cells(i).Value)
            z = SafeText(lo.ListColumns("Zone").DataBodyRange.Cells(i).Value)
            If code <> "" And (zoneFilter = "" Or StrComp(z, zoneFilter, vbTextCompare) = 0) Then
                n = n + 1
                codes(n) = code
                If Len(lst) < 750 Then
                    lst = lst & n & "    " & code & "   " & _
                          SafeText(lo.ListColumns("Description").DataBodyRange.Cells(i).Value) & vbLf
                End If
            End If
        Next i
    End If
    If n = 0 And zoneFilter <> "" Then
        PickItem = PickItem(title & vbLf & "(no items set up for this zone - showing all)", "")
        Exit Function
    End If
    If n = 0 Then
        MsgBox "There are no items yet. Add them on the Stock sheet first.", vbInformation, "Floor Map"
        Exit Function
    End If
    ans = Trim$(InputBox(title & vbLf & vbLf & lst & vbLf & "Type the NUMBER:", "Floor Map"))
    If ans = "" Then Exit Function
    If IsNumeric(ans) Then
        If CLng(ans) >= 1 And CLng(ans) <= n Then
            PickItem = codes(CLng(ans))
            Exit Function
        End If
    End If
    For i = 1 To n
        If StrComp(codes(i), ans, vbTextCompare) = 0 Then
            PickItem = codes(i)
            Exit Function
        End If
    Next i
    MsgBox "'" & ans & "' is not on the list.", vbExclamation, "Floor Map"
End Function

Private Sub ShowInfo(ByVal i As Long, c As Range)
    Dim item As String
    item = CellText(c)
    If item = "" Then
        Application.StatusBar = sqId(i) & "  |  " & ZoneName(sqZone(i)) & "  |  empty"
    Else
        Application.StatusBar = sqId(i) & "  |  " & ZoneName(sqZone(i)) & "  |  " & item & "  " & ItemDesc(item)
    End If
End Sub


'======================================================================
'  SMALL HELPERS
'======================================================================
Private Function SafeText(ByVal v As Variant) As String
    If IsError(v) Then Exit Function
    SafeText = Trim$(CStr(v))
End Function

Private Function CellText(c As Range) As String
    CellText = SafeText(c.Value)
End Function

' The message line under the title (blank text = normal instructions)
Private Sub SetMsg(ByVal text As String, ByVal colour As Long)
    Dim wasBusy As Boolean
    wasBusy = mBusy
    mBusy = True
    If text = "" Then
        Me.Range("A2").Value = "PUT AWAY: click a square " & ChrW(8594) & " arrow " & ChrW(8594) & _
            " pick.     MOVE: double-click a pallet, then click where it goes.     REMOVE: click it, press Delete."
        Me.Range("A2").Font.Color = RGB(230, 0, 126)
    Else
        Me.Range("A2").Value = text
        Me.Range("A2").Font.Color = colour
    End If
    mBusy = wasBusy
    Application.StatusBar = False
End Sub

Private Sub Grid(c As Range)
    Dim e As Variant
    For Each e In Array(xlEdgeLeft, xlEdgeTop, xlEdgeBottom, xlEdgeRight)
        With c.Borders(e)
            .LineStyle = xlContinuous
            .Weight = xlThin
            .Color = vbWhite
        End With
    Next e
End Sub

Private Sub Box(c As Range, ByVal colour As Long)
    Dim e As Variant
    For Each e In Array(xlEdgeLeft, xlEdgeTop, xlEdgeBottom, xlEdgeRight)
        With c.Borders(e)
            .LineStyle = xlContinuous
            .Weight = xlThick
            .Color = colour
        End With
    Next e
End Sub

Private Sub Highlight(c As Range)
    If mHi Is Nothing Then Set mHi = New Collection
    Box c, RGB(230, 0, 126)
    mHi.Add c.Address
End Sub

Private Sub ClearHighlights()
    Dim a As Variant
    If mHi Is Nothing Then Exit Sub
    For Each a In mHi
        Grid Me.Range(a)
    Next a
    Set mHi = New Collection
End Sub

Private Sub PushUndo(ByVal s As String)
    If mUndo Is Nothing Then Set mUndo = New Collection
    mUndo.Add s
    If mUndo.Count > MAX_UNDO Then mUndo.Remove 1
End Sub

Private Sub LogIt(ByVal what As String, ByVal item As String, ByVal fromId As String, ByVal toId As String)
    Dim ws As Worksheet, r As Long
    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets("Moves")
    If ws Is Nothing Then Exit Sub
    r = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row + 1
    If r < 5 Then r = 5
    ws.Cells(r, 1).Value = Now
    ws.Cells(r, 1).NumberFormat = "dd/mm/yyyy hh:mm"
    ws.Cells(r, 2).Value = Environ$("USERNAME")
    ws.Cells(r, 3).Value = what
    ws.Cells(r, 4).Value = item
    ws.Cells(r, 5).Value = fromId
    ws.Cells(r, 6).Value = toId
End Sub

' Draws the five buttons the first time they are needed
Private Sub EnsureButtons()
    Dim s As Shape, caps As Variant, macs As Variant, cols As Variant, i As Long, x As Double
    If Not mStockChecked Then CheckStock
    On Error Resume Next
    Set s = Me.Shapes("fmBtn5")
    On Error GoTo 0
    If Not s Is Nothing Then Exit Sub
    For i = Me.Shapes.Count To 1 Step -1
        If Left$(Me.Shapes(i).Name, 5) = "fmBtn" Then Me.Shapes(i).Delete
    Next i
    caps = Array("Find space", "Find item", "Low stock", "Undo", "Clear", "Rebuild map")
    macs = Array("FindSpace", "FindItem", "LowStock", "UndoLast", "ClearAll", "RebuildMap")
    cols = Array(RGB(230, 0, 126), RGB(230, 0, 126), RGB(192, 0, 0), RGB(64, 64, 64), RGB(128, 128, 128), RGB(0, 112, 192))
    x = Me.Columns(2).Left
    For i = 0 To 5
        Set s = Me.Shapes.AddShape(msoShapeRoundedRectangle, x, Me.Rows(3).Top + 2, 92, Me.Rows(3).Height - 4)
        s.Name = "fmBtn" & i
        s.Fill.ForeColor.RGB = cols(i)
        s.Line.Visible = msoFalse
        s.Placement = xlFreeFloating
        s.OnAction = Me.CodeName & "." & macs(i)
        With s.TextFrame2
            .MarginTop = 0: .MarginBottom = 0
            .VerticalAnchor = msoAnchorMiddle
            .TextRange.Text = caps(i)
            .TextRange.Font.Name = "Arial"
            .TextRange.Font.Size = 10
            .TextRange.Font.Bold = msoTrue
            .TextRange.Font.Fill.ForeColor.RGB = vbWhite
            .TextRange.ParagraphFormat.Alignment = msoAlignCenter
        End With
        x = x + 98
    Next i
    ' A Rebuild map button on the Layout sheet too
    On Error Resume Next
    With ThisWorkbook.Worksheets("Layout")
        .Shapes("fmLayoutBtn").Delete
        Set s = .Shapes.AddShape(msoShapeRoundedRectangle, .Range("I5").Left, .Range("I5").Top, 150, 34)
        s.Name = "fmLayoutBtn"
        s.Fill.ForeColor.RGB = RGB(0, 112, 192)
        s.Line.Visible = msoFalse
        s.OnAction = Me.CodeName & ".RebuildMap"
        With s.TextFrame2
            .VerticalAnchor = msoAnchorMiddle
            .TextRange.Text = "REBUILD MAP"
            .TextRange.Font.Name = "Arial"
            .TextRange.Font.Size = 12
            .TextRange.Font.Bold = msoTrue
            .TextRange.Font.Fill.ForeColor.RGB = vbWhite
            .TextRange.ParagraphFormat.Alignment = msoAlignCenter
        End With
    End With
End Sub


'======================================================================
'  REBUILD MAP  - redraws the Map from the table on the Layout sheet.
'  Pallets stay in their squares (by square name, e.g. L2-05-C).
'  Any that no longer fit are listed and written to the Moves sheet.
'======================================================================

Public Sub RebuildMap()
    Dim lo As ListObject, n As Long, i As Long, j As Long, k As Long, d As Long, t As Long
    Dim aId() As String, aSide() As String, aOrd() As Double, aRows() As Long, aDeep() As Long
    Dim aZone() As String, aBlk() As Boolean, na As Long, errs As String, use As String
    Dim idx() As Long, ni As Long, sd As Long, lw As Long, rw As Long, hL As Long, hR As Long, h As Long
    Dim cur As Long, topR As Long, c0 As Long, c1 As Long, lbl As Range, rng As Range, tl As String
    Dim keepId() As String, keepItem() As String, nk As Long, placed() As Boolean, v As String
    Dim newN As Long, sqs As Worksheet, lost As String, nLost As Long, zi As Long
    Dim zTotal() As Long, zUsed() As String, colour As Long, fc As FormatCondition, m As Long
    Dim curAisle As String

    If MsgBox("Redraw the Map from the Layout sheet?" & vbLf & vbLf & _
              "Pallets stay where they are. Any that no longer fit will be listed.", _
              vbYesNo + vbQuestion, "Rebuild map") = vbNo Then Exit Sub
    On Error GoTo Fail
    CancelMove
    ClearHighlights
    LoadZones

    ' ---- 1. read and check the Layout table
    Set lo = ThisWorkbook.Worksheets("Layout").ListObjects("tblLayout")
    If lo.DataBodyRange Is Nothing Then
        MsgBox "The Layout table is empty.", vbExclamation, "Rebuild map"
        Exit Sub
    End If
    n = lo.ListRows.Count
    ReDim aId(1 To n): ReDim aSide(1 To n): ReDim aOrd(1 To n): ReDim aRows(1 To n)
    ReDim aDeep(1 To n): ReDim aZone(1 To n): ReDim aBlk(1 To n)
    For i = 1 To n
        use = LCase$(LVal(lo, i, "Use"))
        If LVal(lo, i, "Aisle") <> "" And (use = "yes" Or use = "blocked") Then
            na = na + 1
            aId(na) = LVal(lo, i, "Aisle")
            aSide(na) = UCase$(Left$(LVal(lo, i, "Side"), 1))
            aOrd(na) = Val(LVal(lo, i, "Order from fire exit"))
            aRows(na) = CLng(Val(LVal(lo, i, "Rows")))
            aDeep(na) = CLng(Val(LVal(lo, i, "Pallets deep")))
            aZone(na) = UCase$(LVal(lo, i, "Zone"))
            aBlk(na) = (use = "blocked")
            If aSide(na) <> "L" And aSide(na) <> "R" Then errs = errs & vbLf & aId(na) & ": Side must be Left or Right."
            If aRows(na) < 1 Or aRows(na) > 26 Then errs = errs & vbLf & aId(na) & ": Rows must be 1 to 26."
            If aDeep(na) < 1 Or aDeep(na) > 60 Then errs = errs & vbLf & aId(na) & ": Pallets deep must be 1 to 60."
            If Not aBlk(na) And aZone(na) <> "" And Not NameExists("dd_" & aZone(na)) Then
                errs = errs & vbLf & aId(na) & ": zone '" & aZone(na) & "' is not a zone code."
            End If
            For j = 1 To na - 1
                If StrComp(aId(j), aId(na), vbTextCompare) = 0 Then errs = errs & vbLf & aId(na) & " is listed twice."
            Next j
        End If
    Next i
    If na = 0 Then errs = errs & vbLf & "No aisles have Use = Yes."
    If errs <> "" Then
        MsgBox "Please fix the Layout sheet first:" & vbLf & errs, vbExclamation, "Rebuild map"
        Exit Sub
    End If

    ' ---- 2. remember what is on the map now
    LoadSquares
    ReDim keepId(1 To sqN + 1): ReDim keepItem(1 To sqN + 1)
    For i = 1 To sqN
        v = CellText(Me.Cells(sqRow(i), sqCol(i)))
        If v <> "" Then
            nk = nk + 1
            keepId(nk) = sqId(i)
            keepItem(nk) = v
        End If
    Next i
    ReDim placed(1 To nk + 1)

    ' ---- 3. work out the size of everything (fire exit at the bottom, bay at the top)
    For i = 1 To na
        If aSide(i) = "L" Then
            If aDeep(i) > lw Then lw = aDeep(i)
            hL = hL + 1 + aRows(i)
        Else
            If aDeep(i) > rw Then rw = aDeep(i)
            hR = hR + 1 + aRows(i)
        End If
    Next i
    h = hL: If hR > h Then h = hR
    If 15 + zN > h Then h = 15 + zN
    gBottom = TOP_ROW + h - 1
    gWalkS = 2 + lw
    gWalkE = gWalkS + WALK - 1
    gRightS = gWalkE + 1
    gMapLast = gWalkE: If gRightS + rw - 1 > gMapLast Then gMapLast = gRightS + rw - 1
    gLegend = gMapLast + 2
    gLast = gLegend + LEG_COLS - 1
    gFire = gBottom + 2

    ' ---- 4. wipe the old drawing
    mBusy = True
    Application.EnableEvents = False
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Me.Activate
    With Me.Cells
        .UnMerge
        .Validation.Delete
        .FormatConditions.Delete
        .ClearComments
        .Clear
        .ColumnWidth = Me.StandardWidth
        .RowHeight = Me.StandardHeight
    End With
    Set mUndo = New Collection

    ' square cells
    Me.Columns(1).ColumnWidth = SQ_PX / 7
    For k = 1 To 3
        Me.Columns(1).ColumnWidth = Me.Columns(1).ColumnWidth * (SQ_PX * 0.75) / Me.Columns(1).Width
    Next k
    Me.Range(Me.Columns(1), Me.Columns(gLast)).ColumnWidth = Me.Columns(1).ColumnWidth
    Me.Rows(1).RowHeight = 30
    Me.Rows(2).RowHeight = 22
    Me.Rows(3).RowHeight = 24
    Me.Range(Me.Rows(4), Me.Rows(gFire)).RowHeight = SQ_PX * 0.75

    ' ---- 5. title, message line, bay, fire exit, walkway
    With Me.Range(Me.Cells(1, 1), Me.Cells(1, gLast))
        .Merge
        .Cells(1, 1).Value = "Consumables Floor Map, Sherburn"
        .Font.Name = "Arial": .Font.Size = 18: .Font.Bold = True: .Font.Color = vbWhite
        .Interior.Color = vbBlack
        .VerticalAlignment = xlCenter
        .IndentLevel = 1
        With .Borders(xlEdgeBottom)
            .LineStyle = xlContinuous: .Weight = xlThick: .Color = RGB(230, 0, 126)
        End With
    End With
    With Me.Range(Me.Cells(2, 1), Me.Cells(2, gLast))
        .Merge
        .Font.Name = "Arial": .Font.Size = 12: .Font.Bold = True
        .VerticalAlignment = xlCenter
        .IndentLevel = 1
    End With
    SetMsg "", 0
    mBusy = True
    Bar 4, "LOADING BAY", vbBlack
    Bar gFire, "FIRE EXIT", RGB(0, 132, 61)
    With Me.Range(Me.Cells(TOP_ROW, gWalkS), Me.Cells(gBottom, gWalkE))
        .Merge
        .Cells(1, 1).Value = "WALKWAY"
        .Orientation = 90
        .Font.Name = "Arial": .Font.Size = 12: .Font.Bold = True: .Font.Color = RGB(191, 191, 191)
        .HorizontalAlignment = xlCenter
        .VerticalAlignment = xlCenter
    End With

    ' ---- 6. aisles, stacked up from the fire exit on each side
    ReDim zTotal(1 To zN + 1): ReDim zUsed(1 To zN + 1)
    newN = 0
    Set sqs = ThisWorkbook.Worksheets("Squares")
    sqs.Cells.Clear
    sqs.Range("A1:F1").Value = Array("ID", "Aisle", "Zone", "Row", "Col", "Depth")
    For sd = 1 To 2
        ' aisles on this side, sorted by Order
        ReDim idx(1 To na): ni = 0
        For i = 1 To na
            If (sd = 1 And aSide(i) = "L") Or (sd = 2 And aSide(i) = "R") Then
                ni = ni + 1: idx(ni) = i
            End If
        Next i
        For i = 2 To ni
            t = idx(i): j = i - 1
            Do While j >= 1
                If aOrd(idx(j)) <= aOrd(t) Then Exit Do
                idx(j + 1) = idx(j): j = j - 1
            Loop
            idx(j + 1) = t
        Next i

        cur = gBottom
        For k = 1 To ni
            i = idx(k)
            curAisle = aId(i)
            topR = cur - aRows(i) + 1
            If sd = 1 Then c0 = gWalkS - aDeep(i): c1 = gWalkS - 1 Else c0 = gRightS: c1 = gRightS + aDeep(i) - 1

            ' label above the aisle
            Set lbl = Me.Range(Me.Cells(topR - 1, c0), Me.Cells(topR - 1, c1))
            If c1 > c0 Then lbl.Merge
            lbl.Font.Name = "Arial": lbl.Font.Size = 11: lbl.Font.Bold = True
            lbl.HorizontalAlignment = xlLeft
            lbl.VerticalAlignment = xlBottom
            If aBlk(i) Then
                lbl.Cells(1, 1).Value = aId(i) & "   NOT OURS"
            ElseIf aZone(i) <> "" Then
                lbl.Cells(1, 1).Formula = "=""" & Replace(aId(i), """", """""") & "   ""&zn_" & aZone(i)
            Else
                lbl.Cells(1, 1).Value = aId(i) & "   No zone"
            End If

            Set rng = Me.Range(Me.Cells(topR, c0), Me.Cells(cur, c1))
            If aBlk(i) Then
                ' someone else's space
                rng.Merge
                rng.Cells(1, 1).Value = "NOT OURS - someone else's space"
                rng.Interior.Color = RGB(191, 191, 191)
                rng.Font.Name = "Arial": rng.Font.Size = 11: rng.Font.Bold = True
                rng.Font.Color = RGB(128, 128, 128)
                rng.HorizontalAlignment = xlCenter
                rng.VerticalAlignment = xlCenter
            Else
                zi = ZoneIdx(aZone(i))
                If zi > 0 Then colour = ZoneColour(aZone(i)) Else colour = RGB(217, 217, 217)
                ' stripes = free space
                With rng.Interior
                    If zi > 0 Then
                        .Pattern = xlPatternLightUp
                        .Color = vbWhite
                        .PatternColor = colour
                    Else
                        .Pattern = xlPatternSolid
                        .Color = colour
                    End If
                End With
                rng.Font.Name = "Arial": rng.Font.Size = 7: rng.Font.Bold = True
                rng.HorizontalAlignment = xlCenter
                rng.VerticalAlignment = xlCenter
                rng.ShrinkToFit = True
                rng.Borders.LineStyle = xlContinuous
                rng.Borders.Weight = xlThin
                rng.Borders.Color = vbWhite
                ' dropdown of this zone's items
                With rng.Validation
                    .Add Type:=xlValidateList, AlertStyle:=xlValidAlertWarning, _
                         Formula1:="=dd_" & IIf(zi > 0, aZone(i), "ALL")
                    .IgnoreBlank = True
                    .InCellDropdown = True
                    .ErrorTitle = "Not for this zone"
                    .ErrorMessage = "That item isn't listed for this zone (or isn't on the Stock sheet). Put it here anyway?"
                    .InputTitle = Left$("Aisle " & aId(i), 32)      ' Excel allows 32 characters here
                    .InputMessage = "Click the arrow and pick an item. Press Delete to empty."
                    .ShowInput = True
                    .ShowError = True
                End With
                ' colours when full (relative formulas follow the active cell)
                tl = rng.Cells(1, 1).Address(False, False)
                rng.Cells(1, 1).Select
                If zi > 0 Then
                    Set fc = rng.FormatConditions.Add(Type:=xlExpression, Formula1:="=AND(LEN(" & tl & _
                        ")>0,IFERROR(INDEX(ItemZones,MATCH(" & tl & ",ItemCodes,0)),""" & aZone(i) & """)<>""" & aZone(i) & """)")
                    fc.Interior.Color = RGB(255, 0, 0)
                    fc.Font.Color = vbWhite
                    fc.StopIfTrue = True
                    zTotal(zi) = zTotal(zi) + aRows(i) * aDeep(i)
                    zUsed(zi) = zUsed(zi) & "+COUNTA(Map!" & rng.Address & ")"
                End If
                Set fc = rng.FormatConditions.Add(Type:=xlExpression, Formula1:="=LEN(" & tl & ")>0")
                If zi > 0 Then fc.Interior.Color = colour Else fc.Interior.Color = RGB(166, 166, 166)
                If IsDark(IIf(zi > 0, colour, RGB(166, 166, 166))) Then fc.Font.Color = vbWhite Else fc.Font.Color = vbBlack

                ' the squares themselves (+ put pallets back)
                For t = 0 To aRows(i) - 1
                    For d = 1 To aDeep(i)
                        newN = newN + 1
                        v = aId(i) & "-" & Format$(d, "00") & "-" & Chr$(65 + t)
                        If sd = 1 Then j = gWalkS - d Else j = gRightS + d - 1
                        sqs.Cells(newN + 1, 1).Resize(1, 6).Value = Array(v, aId(i), IIf(zi > 0, aZone(i), ""), topR + t, j, d)
                        For m = 1 To nk
                            If Not placed(m) Then
                                If StrComp(keepId(m), v, vbTextCompare) = 0 Then
                                    Me.Cells(topR + t, j).Value = keepItem(m)
                                    placed(m) = True
                                    Exit For
                                End If
                            End If
                        Next m
                    Next d
                Next t
            End If
            cur = topR - 2
        Next k
    Next sd

    ' ---- 7. legend, free spaces and Stock sheet counts
    DrawLegend zTotal, zUsed
    For i = 1 To nk
        If Not placed(i) Then
            nLost = nLost + 1
            If nLost <= 20 Then lost = lost & vbLf & "   " & keepId(i) & ":  " & keepItem(i)
            LogIt "Removed", keepItem(i), keepId(i), "(no longer on the map)"
        End If
    Next i

    Me.PageSetup.PrintArea = Me.Range(Me.Cells(1, 1), Me.Cells(gFire, gLast)).Address
    sqN = 0                                  ' re-read squares next time
    RepairStock
    Application.Calculation = xlCalculationAutomatic
    Application.ScreenUpdating = True
    Me.Range(Me.Cells(1, 1), Me.Cells(gFire, gLast)).Select
    ActiveWindow.Zoom = True
    If ActiveWindow.Zoom > 100 Then ActiveWindow.Zoom = 100
    Me.Range("A1").Select
    Application.EnableEvents = True
    mBusy = False
    MsgBox "Map rebuilt: " & newN & " squares." & IIf(nLost > 0, vbLf & vbLf & nLost & _
           " pallet(s) no longer fit (also written on the Moves sheet):" & lost, ""), _
           IIf(nLost > 0, vbExclamation, vbInformation), "Rebuild map"
    Exit Sub
Fail:
    Dim failMsg As String
    failMsg = "Rebuild stopped" & IIf(curAisle <> "", " while drawing aisle '" & curAisle & "'", "") & _
              ":" & vbLf & Err.Description
    On Error Resume Next
    ' Nothing is lost: pallets not yet put back are written on the Moves sheet
    For m = 1 To nk
        If Not placed(m) Then LogIt "Not redrawn", keepItem(m), keepId(m), "(rebuild stopped - put back by hand)"
    Next m
    sqN = 0
    mBusy = False
    Application.Calculation = xlCalculationAutomatic
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    MsgBox failMsg & vbLf & vbLf & "Any pallets that were not put back are listed on the Moves sheet. " & _
           "Fix the Layout sheet and press Rebuild map again.", vbExclamation, "Rebuild map"
End Sub

' Once per session: if the Stock count formula is old or was changed, fix it
Private Sub CheckStock()
    Dim lo As ListObject
    mStockChecked = True
    On Error Resume Next
    Set lo = ThisWorkbook.Worksheets("Stock").ListObjects("tblStock")
    If lo Is Nothing Then Exit Sub
    If lo.DataBodyRange Is Nothing Then Exit Sub
    If InStr(lo.ListColumns("Pallets on map").DataBodyRange.Cells(1, 1).Formula, "$ZZ$2000") = 0 Then
        RepairStock
    End If
End Sub

' Makes sure every item on the Stock sheet is inside the table and has its
' "Pallets on map" and "Status" formulas (fixes items typed under the table)
Private Sub RepairStock()
    Dim ws As Worksheet, lo As ListObject, lastRow As Long, want As Long, top As Range
    Set ws = ThisWorkbook.Worksheets("Stock")
    Set lo = ws.ListObjects("tblStock")
    Set top = lo.HeaderRowRange.Cells(1, 1)
    lastRow = ws.Cells(ws.Rows.Count, top.Column).End(xlUp).Row
    want = top.Row + 150
    If lastRow > want Then want = lastRow
    If top.Row + lo.ListRows.Count < want Then
        lo.Resize ws.Range(top, ws.Cells(want, top.Column + lo.ListColumns.Count - 1))
    End If
    lo.ListColumns("Pallets on map").DataBodyRange.Formula = COUNT_FORMULA
    lo.ListColumns("Status").DataBodyRange.Formula = _
        "=IF(TRIM([@[Item code]])="""","""",IF([@[Pallets on map]]<=IF([@[Low when pallets at or below]]="""",2," & _
        "[@[Low when pallets at or below]]),""LOW"",""OK""))"
End Sub

Private Sub DrawLegend(zTotal() As Long, zUsed() As String)
    Dim r As Long, z As Long, lc As Long, keys As Variant, k As Long, st As Worksheet, nm As Range
    lc = gLegend
    r = TOP_ROW
    LegBand r, "COLOURS"
    For z = 1 To zN
        r = r + 1
        Me.Cells(r, lc).Interior.Color = ZoneColour(zCode(z))
        LegText r, "=zn_" & zCode(z)
    Next z
    keys = Array("Stripes = free space", "Red = wrong zone, move it", "Grey = not ours")
    For k = 0 To 2
        r = r + 1
        Select Case k
            Case 0
                Me.Cells(r, lc).Interior.Pattern = xlPatternLightUp
                Me.Cells(r, lc).Interior.Color = vbWhite
                Me.Cells(r, lc).Interior.PatternColor = ZoneColour(zCode(1))
            Case 1: Me.Cells(r, lc).Interior.Color = RGB(255, 0, 0)
            Case 2: Me.Cells(r, lc).Interior.Color = RGB(191, 191, 191)
        End Select
        LegText r, CStr(keys(k))
    Next k
    r = r + 2
    LegBand r, "FREE SPACES"
    Set st = ThisWorkbook.Worksheets("Stock")
    For z = 1 To zN
        Set nm = ThisWorkbook.Names("zn_" & zCode(z)).RefersToRange
        If zTotal(z) > 0 Then
            r = r + 1
            With Me.Range(Me.Cells(r, lc), Me.Cells(r, lc + 1))
                .Merge
                .Cells(1, 1).Formula = "=" & zTotal(z) & "-(" & Mid$(Replace(zUsed(z), "Map!", ""), 2) & ")"
                .Font.Name = "Arial": .Font.Size = 12: .Font.Bold = True: .Font.Color = RGB(230, 0, 126)
                .HorizontalAlignment = xlRight
            End With
            LegText r, "=zn_" & zCode(z)
            nm.Offset(0, 1).Value = zTotal(z)
            nm.Offset(0, 2).Formula = "=" & Mid$(zUsed(z), 2)
        Else
            nm.Offset(0, 1).Value = 0
            nm.Offset(0, 2).Value = 0
        End If
        nm.Offset(0, 3).Formula = "=" & nm.Offset(0, 1).Address(False, False) & "-" & nm.Offset(0, 2).Address(False, False)
    Next z
    r = r + 2
    LegBand r, "LOW STOCK"
    r = r + 1
    With Me.Range(Me.Cells(r, lc), Me.Cells(r, lc + 1))
        .Merge
        .Cells(1, 1).Formula = "=COUNTIF(tblStock[Status],""LOW"")"
        .Font.Name = "Arial": .Font.Size = 12: .Font.Bold = True: .Font.Color = RGB(192, 0, 0)
        .HorizontalAlignment = xlRight
    End With
    LegText r, "items LOW - see Stock sheet"
End Sub

Private Sub LegBand(ByVal r As Long, ByVal text As String)
    With Me.Range(Me.Cells(r, gLegend), Me.Cells(r, gLegend + LEG_COLS - 1))
        .Merge
        .Cells(1, 1).Value = text
        .Font.Name = "Arial": .Font.Size = 11: .Font.Bold = True: .Font.Color = vbWhite
        .Interior.Color = vbBlack
        .VerticalAlignment = xlCenter
        .IndentLevel = 1
    End With
End Sub

Private Sub LegText(ByVal r As Long, ByVal text As String)
    With Me.Range(Me.Cells(r, gLegend + 2), Me.Cells(r, gLegend + LEG_COLS - 1))
        .Merge
        If Left$(text, 1) = "=" Then .Cells(1, 1).Formula = text Else .Cells(1, 1).Value = text
        .Font.Name = "Arial": .Font.Size = 11
        .VerticalAlignment = xlCenter
    End With
End Sub

Private Sub Bar(ByVal r As Long, ByVal text As String, ByVal colour As Long)
    With Me.Range(Me.Cells(r, 2), Me.Cells(r, gMapLast))
        .Merge
        .Cells(1, 1).Value = text
        .Interior.Color = colour
        .Font.Name = "Arial": .Font.Size = 12: .Font.Bold = True: .Font.Color = vbWhite
        .HorizontalAlignment = xlCenter
        .VerticalAlignment = xlCenter
    End With
End Sub

' Zone codes in the order they appear on the Stock sheet (names zn_CODE)
Private Sub LoadZones()
    Dim nm As Name, codes() As String, rows() As Long, i As Long, j As Long, tc As String, tr As Long
    ReDim codes(1 To ThisWorkbook.Names.Count + 1): ReDim rows(1 To ThisWorkbook.Names.Count + 1)
    zN = 0
    For Each nm In ThisWorkbook.Names
        If LCase$(Left$(nm.Name, 3)) = "zn_" Then
            zN = zN + 1
            codes(zN) = Mid$(nm.Name, 4)
            rows(zN) = nm.RefersToRange.Row
        End If
    Next nm
    For i = 2 To zN
        tc = codes(i): tr = rows(i): j = i - 1
        Do While j >= 1
            If rows(j) <= tr Then Exit Do
            codes(j + 1) = codes(j): rows(j + 1) = rows(j): j = j - 1
        Loop
        codes(j + 1) = tc: rows(j + 1) = tr
    Next i
    ReDim zCode(1 To zN + 1)
    For i = 1 To zN
        zCode(i) = UCase$(codes(i))
    Next i
End Sub

Private Function ZoneIdx(ByVal code As String) As Long
    Dim i As Long
    For i = 1 To zN
        If StrComp(zCode(i), code, vbTextCompare) = 0 Then ZoneIdx = i: Exit Function
    Next i
End Function

' Zone colour = the fill of its Code cell on the Stock sheet
Private Function ZoneColour(ByVal code As String) As Long
    On Error Resume Next
    ZoneColour = RGB(217, 217, 217)
    ZoneColour = ThisWorkbook.Names("zn_" & code).RefersToRange.Offset(0, -1).Interior.Color
End Function

Private Function IsDark(ByVal c As Long) As Boolean
    IsDark = (0.299 * (c And &HFF&) + 0.587 * ((c \ &H100&) And &HFF&) + 0.114 * ((c \ &H10000) And &HFF&)) < 140
End Function

Private Function NameExists(ByVal nm As String) As Boolean
    Dim x As Name
    On Error Resume Next
    Set x = ThisWorkbook.Names(nm)
    NameExists = Not x Is Nothing
End Function

Private Function LVal(lo As ListObject, ByVal i As Long, ByVal col As String) As String
    LVal = SafeText(lo.ListColumns(col).DataBodyRange.Cells(i).Value)
End Function
